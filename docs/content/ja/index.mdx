let res = list_inbox_emails(&client, ListInboxEmailsParams {

<Section
  half
  kind="言語"
  title="3つの言語に対応し、依存は最低限"
  lede="ランタイムはライブラリとして配られるのではなく、リクエストと一緒に生成されます。プロジェクトが抱えるのは生成されたディレクトリだけで、そのほかはほとんどありません。"
>

<Traits>
  <Trait title="Go" lang="go">
    生成されたパッケージは標準ライブラリしか使いません。通信はすでに手元にある
    `http.Client`です。
  </Trait>
  <Trait title="Rust" lang="rust">
    `serde`と`serde_json`だけです。HTTPスタックもTLSバックエンドもasync
    ランタイムも付いてきません。通信は、プログラムがすでに使っているクライアント
    の上に実装する`Transport`です。
  </Trait>
  <Trait title="TypeScript" lang="typescript">
    依存はありません。プラットフォームに求めるのは`fetch`だけです。
  </Trait>
</Traits>

<Section
  half
  kind="対応範囲"
  title="JMAPが標準化したすべて"
  lede="デモが動く範囲を選んだものではありません。標準がここまでに公開したプロトコルそのものです。"
>

<Traits>
  <Trait title="15のケイパビリティ">
    IANAが登録しているものすべて（メール、送信、連絡先、カレンダー、
    principals、Sieve、クォータ、blob、プッシュ、そのほか）が、リクエストを
    突き合わせるデータモデルに入っています。載っていないもの、たとえば
    ベンダー独自のものは、スキーマファイルに書けば同じように検証されます。
  </Trait>
  <Trait title="81のメソッド">
    21の型にまたがり、どれも同じように生成され、検証されます。`Email/query`
    も`CalendarEvent/copy`も`SieveScript/validate`も、jmapcにとっては
    同じ種類のものです。
  </Trait>
</Traits>

<Section
  invert
  title="ラッパーではなく、コンパイラであること"
  lede="次の5つは、どれもこの違いから導かれます。"
>

<Traits>
  <Trait title="独自のAPIがない">
    リクエストファイルはJMAPそのものです。覚えるのは仕様であり、3つの言語で
    同じ知識が通用します。
  </Trait>
  <Trait title="訊いたとおりの形で返る">
    結果はリクエストが要求したプロパティだけを持ちます。ひとつ増やせば型が育ち、
    存在しないものを要求すればビルドが止まります。
  </Trait>
  <Trait title="JMAPが報告するすべての層のエラー">
    リクエスト全体の失敗、メソッド呼び出しの失敗、200を返しながら拒否している
    /set。生成されたコードがこれらを区別するので、呼び出し側は区別せずに済みます。
  </Trait>
  <Trait title="プッシュとページングはループとして">
    watchを指定したリクエストは、サーバが報告する変更を追い続けます。pagedを
    指定したリクエストは、一部ずつ返る結果を最後まで読み通します。
  </Trait>
  <Trait title="まわりの道具">
    実際のサーバにリクエストを送るコマンド、自分のコードをテストするためのJMAP
    サーバ、そしてjmapcがまだ知らないケイパビリティを書くスキーマファイル。
  </Trait>
</Traits>
